Table PACIENTE {
  ID bigint [pk, increment]
  FICHA_DOCUMENTO_ID bigint [ref: > DOCUMENTO.ID]
  NOME varchar(400) [not null]
  APELIDO varchar(400)
  CPF varchar(40)
  RG varchar(40)
  TELEFONE varchar(40)
  ENDERECO_RUA varchar(255)
  ENDERECO_NR varchar(40)
  ENDERECO_BAIRRO varchar(255)
  ENDERECO_MUNICIPIO varchar(255)
  ENDERECO_ESTADO varchar(255)
  DT_NASC date
  ATIVO boolean
  DT_CRIACAO timestamp
  DT_DESATIVADO timestamp
}

Table TIPO_DEPENDENCIA {
  ID bigint [pk, increment]
  NOME varchar(200) [not null]
  CATEGORIA int [not null]
}

Table RESPONSAVEL {
  ID bigint [pk, increment]
  PACIENTE_ID bigint [not null, ref: > PACIENTE.ID]
  NOME varchar(400)
  PARENTESCO varchar(40)
  CPF varchar(40)
  RG varchar(40)
  TELEFONE varchar(40)
  ENDERECO_RUA varchar(255)
  ENDERECO_NR varchar(40)
  ENDERECO_BAIRRO varchar(255)
  ENDERECO_MUNICIPIO varchar(255)
  ENDERECO_ESTADO varchar(255)
  DT_NASC date
}

Table TIPO_DOCUMENTO {
  ID bigint [pk, increment]
  NOME varchar(200) [not null]
  DESCRICAO text [not null]
  EXIGE_ASSINATURA boolean
  ATIVO boolean
  DT_DESATIVADO timestamp
}

Table TIPO_DOCUMENTO_VERSAO {
  ID bigint [pk, increment]
  TIPO_DOCUMENTO_ID bigint [not null, ref: > TIPO_DOCUMENTO.ID]
  VERSAO int
  CAMPOS jsonb
  TEMPLATE varchar(400)
  DT_CRIACAO timestamp
}

Table DOCUMENTO {
  ID bigint [pk, increment]
  TIPO_DOCUMENTO_VERSAO_ID bigint [not null, ref: > TIPO_DOCUMENTO_VERSAO.ID]
  DOCUMENTO_ORIGEM_ID bigint [ref: > DOCUMENTO.ID]
  PACIENTE_ID bigint [not null, ref: > PACIENTE.ID]
  USUARIO_ID bigint [ref: > USUARIO.ID]
  OBSERVACAO text
  DADOS jsonb
  STATUS varchar(20) [not null, default: 'PENDENTE']
  HASH varchar(64)
  CAMINHO_ASSINADO varchar(400)
  ATIVO boolean
  DT_CRIACAO timestamp
  DT_ATUALIZACAO timestamp
  DT_ASSINATURA timestamp
  DT_FINALIZACAO timestamp
  DT_DESATIVADO timestamp
}

Table PROFISSIONAL {
  ID bigint [pk, increment]
  NOME varchar(255) [not null]
  TELEFONE varchar(40)
  REG_PROFISSIONAL varchar(255)
  CPF varchar(40)
  ENDERECO_RUA varchar(255)
  ENDERECO_NR varchar(40)
  ENDERECO_BAIRRO varchar(255)
  ENDERECO_MUNICIPIO varchar(255)
  ENDERECO_ESTADO varchar(255)
  DT_NASC date
}

Table USUARIO {
  ID bigint [pk, increment]
  PERFIL_ID bigint [ref: > PERFIL.ID]
  PROFISSIONAL_ID bigint [unique, ref: - PROFISSIONAL.ID]
  EMAIL varchar(255) [not null, unique]
  SENHA varchar(255) [not null]
  ROLE varchar(40)
  ATIVO boolean [default: true]
  DT_DESATIVADO timestamp
  DT_CRIACAO timestamp
}

Table TOKEN {
  ID bigint [pk, increment]
  USUARIO_ID bigint [ref: > USUARIO.ID]
  TOKEN varchar(255)
}

Table PERFIL {
  ID bigint [pk, increment]
  NOME varchar(255) [not null]
  PERM_CAD_PACIENTE boolean [default: false]
}

Table PERMISSAO_TIPO_DOCUMENTO {
  ID bigint [pk, increment]
  PERFIL_ID bigint [not null, ref: > PERFIL.ID]
  TIPO_DOCUMENTO_ID bigint [not null, ref: > TIPO_DOCUMENTO.ID]
}

Table AVISO {
  ID bigint [pk, increment]
  USUARIO_ID bigint [not null, ref: > USUARIO.ID]
  MENSAGEM text [not null]
  DT_CRIACAO timestamp [default: `CURRENT_TIMESTAMP`]
  DT_FINALIZACAO timestamp
  ATIVO boolean
  DT_DESATIVADO timestamp
}

Table HIST_DOCUMENTO {
  ID bigint [pk, increment]
  USUARIO_ID bigint [ref: > USUARIO.ID]
  DOCUMENTO_ID bigint [ref: > DOCUMENTO.ID]
  PACIENTE_ID bigint [ref: > PACIENTE.ID]
  ACAO varchar(200)
  OBSERVACAO text
  DT_CRIACAO timestamp
}
